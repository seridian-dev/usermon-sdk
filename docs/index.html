Telemetry Ingestion Architecture

Usermon uses a batched, micro-chunked telemetry pipeline that guarantees minimal memory footprint and zero main-thread blockage across client browsers, server runtimes, and mobile devices:

SDK Clients Browser / Mobile / Server / CLI
── [W3C traceparent] ──▶
HTTP Ingest Endpoint POST /v1/ingest (Convex site)
── [Batched Mutation] ──▶
Convex Storage & AI Fingerprinting & Rollups

import usermon

client = usermon.init(

    platform="web",

Install via go get go get github.com/usermon/usermon-sdk-go 
package main

import (
    "context"
    "github.com/usermon/usermon-sdk-go/usermon"
)

func main() {
    client := usermon.New(
        "https://xxx.convex.site",
        "um_YOUR_PROJECT_KEY",
        usermon.WithPlatform(usermon.PlatformWeb),
        usermon.WithRelease("1.0.0"),
    )
    ctx := context.Background()

    client.CaptureException(ctx, "SQL timeout", usermon.ExceptionOpts{
        Route: usermon.Ptr("/api/users"),
    })

    client.CaptureSpan(ctx, usermon.SpanOpts{
        Method: "POST", Route: "/api/orders", Status: 201, DurationMs: 142,
    })
}

# Set environment variables
export USERMON_INGEST_URL=https://xxx.convex.site
export USERMON_INGEST_KEY=um_YOUR_PROJECT_KEY

# Health check
usermon health

# Send events
usermon send exception -m "Crash in payment worker" --level fatal --route /workers

usermon send log --level warn -m "Disk at 88%" --attrs '{"disk":"/dev/sda1"}'
